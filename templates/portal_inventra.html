Relational Database Architecture · SQL & Schema

INVENTRA Transactional Integrity Schema 

-- INVENTRA Enterprise Schema: Stock & Transaction Integrity
CREATE TABLE products (
    sku VARCHAR(24) PRIMARY KEY,
    name VARCHAR(120) NOT NULL,
    category VARCHAR(60) NOT NULL,
    unit_price DECIMAL(10, 2) NOT NULL,
    stock_quantity INT NOT NULL DEFAULT 0,
    reorder_threshold INT NOT NULL DEFAULT 15,
    supplier_id INT REFERENCES suppliers(id)
);

CREATE TABLE stock_audit_log (
    audit_id SERIAL PRIMARY KEY,
    sku VARCHAR(24) REFERENCES products(sku),
    quantity_delta INT NOT NULL,
    reason VARCHAR(50) NOT NULL,  -- 'DISPATCH', 'RESTOCK', 'SUPPLIER_RETURN'
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
